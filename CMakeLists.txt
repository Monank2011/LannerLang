option(LANNER_ENABLE_LEGACY_HIR "Enable the legacy HIR/SSA backend for bootstrap diagnostics" OFF)

    src/lexer/lexer.cpp
    src/parser/parser.cpp

)

    target_compile_options(lanner_frontend PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(lanner_frontend PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(lanner
    src/main.cpp
    src/driver/compiler_driver.cpp

    src/driver/compiler_driver_game.cpp

)

if(MSVC)
    target_compile_options(lanner PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(lanner PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

target_compile_definitions(lanner PRIVATE LANNER_CLANG_EXECUTABLE=\"${LANNER_CLANG_EXECUTABLE}\")
target_compile_definitions(lanner PRIVATE LANNER_RUNTIME_SOURCE=\"${CMAKE_CURRENT_SOURCE_DIR}/src/runtime/lanner_runtime.c\")
